#include <iostream>
#include <string>
#include <limits>

using namespace std;

struct Nod {
	string titlu;
	string artist;
	double minute;
	bool liked;
	Nod* prev;
	Nod* next;
};

void wipe() {
	system("cls");
}

Nod* creeazaNod(string t, string a, double m, bool l) {
	Nod* p = new Nod;
	p->titlu = t;
	p->artist = a;
	p->minute = m;
	p->liked = l;
	p->prev = nullptr;
	p->next = nullptr;
	return p;
}

void afiseaza(Nod* prim) {
	if (prim == nullptr) {
		cout << "Lista este goala.\n";
		return;
	}

	Nod* p = prim;
	while (p != nullptr) {
		cout << "- " << p->titlu << " | " << p->artist << " | " << p->minute << " min | ";
		if (p->liked) cout << "liked - da\n";
		else cout << "liked - nu\n";
		p = p->next;
	}
	cout << "\n";
}

void adauga(Nod*& prim, Nod*& ultim, string t, string a, double m, bool l) {
	Nod* nou = creeazaNod(t, a, m, l);
	if (prim == nullptr) {
		prim = ultim = nou;
		return;
	}
	nou->next = prim;
	prim->prev = nou;
	prim = nou;
}

Nod* cautaT(Nod* prim, string t) {
	Nod* p = prim;
	while (p != nullptr) {
		if (p->titlu == t) return p;
		p = p->next;
	}
	return nullptr;
}

Nod* cautaA(Nod* prim, string a) {
	Nod* p = prim;
	while (p != nullptr) {
		if (p->artist == a) return p;
		p = p->next;
	}
	return nullptr;
}

void stergeNod(Nod*& prim, Nod*& ultim, Nod* q) {
	if (q == nullptr) return;

	if (q == prim && q == ultim) {
		prim = ultim = nullptr;
	}
	else if (q == prim) {
		prim = prim->next;
		prim->prev = nullptr;
	}
	else if (q == ultim) {
		ultim = ultim->prev;
		ultim->next = nullptr;
	}
	else {
		Nod* stanga = q->prev;
		Nod* dreapta = q->next;
		stanga->next = dreapta;
		dreapta->prev = stanga;
	}
	delete q;
}

bool stergeT(Nod*& prim, Nod*& ultim, string t) {
	Nod* q = cautaT(prim, t);
	if (!q) return false;
	stergeNod(prim, ultim, q);
	return true;
}

bool stergeA(Nod*& prim, Nod*& ultim, string a) {
	Nod* p = prim;
	bool sters = false;

	while (p != nullptr) {
		Nod* urmator = p->next;
		if (p->artist == a) {
			stergeNod(prim, ultim, p);
			sters = true;
		}
		p = urmator;
	}
	return sters;
}

void goleste(Nod*& prim, Nod*& ultim) {
	while (prim != nullptr) {
		stergeNod(prim, ultim, prim);
	}
}

void cautarePieseArtist(Nod* prim, string a) {
	Nod* p = prim;
	bool gasit = false;

	while (p != nullptr) {
		if (p->artist == a) {
			if (!gasit) {
				cout << "Piesele artistului " << a << ":\n";
				gasit = true;
			}
			cout << "-> " << p->titlu << " - " << p->minute << " min | ";
			if (p->liked) cout << "liked - da\n";
			else cout << "liked - nu\n";
		}
		p = p->next;
	}

	if (!gasit) {
		cout << "Nu a fost gasita nicio piesa pentru artistul '" << a << "'.\n";
	}
}

double calcTimp(Nod* prim) {
	double s = 0;
	Nod* p = prim;
	while (p != nullptr) {
		s += p->minute;
		p = p->next;
	}
	return s;
}

void meniu() {
	cout << "1.Adauga piesa.\n"
		<< "2.Cauta dupa titlu.\n"
		<< "3.Cauta dupa artist.\n"
		<< "4.Sterge piesa dupa titlu.\n"
		<< "5.Sterge toate piesele de la artist.\n"
		<< "6.Adauga o piesa la liked.\n"
		<< "7.Afiseaza durata playlistului.\n"
		<< "8.Afiseaza.\n"
		<< "9.Goleste lista.\n"
		<< "0.Iesire.\n";
}

int main() {
	Nod* prim = nullptr;
	Nod* ultim = nullptr;

	string titlu, artist;
	double minute;
	bool liked = false;
	int optiune = 0;

	do {
		meniu();
		cout << "\n-Optiune : ";
		cin >> optiune;
		cin.ignore();
		wipe();

		switch (optiune) {
		case 1:
			cout << "Introduceti piesa noua :\n";
			cout << "->Titlu: "; getline(cin, titlu);
			cout << "->Artist: "; getline(cin, artist);
			while (true) {
				cout << "->Minute: ";
				cin >> minute;
				if (cin.fail() || minute < 0) {
					cout << "Minutele trebuie sa fie numere pozitive\n";
					cin.clear();
					cin.ignore(numeric_limits<streamsize>::max(), '\n');
				}
				else {
					cin.ignore(numeric_limits<streamsize>::max(), '\n');
					break;
				}
			}
			adauga(prim, ultim, titlu, artist, minute, liked);
			wipe();
			break;

		case 2:
			if (prim == nullptr) cout << "Nu exista piese inca.\n";
			else {
				cout << "Ce piesa trebuie cautata? ";
				getline(cin, titlu);
				if (cautaT(prim, titlu) != nullptr)
					cout << "Piesa '" << titlu << "' a fost gasita.\n";
				else
					cout << "Nu exista piesa '" << titlu << "'.\n";
			}
			break;

		case 3:
			if (prim == nullptr) cout << "Nu exista inca artisti.\n";
			else {
				cout << "Ce artist trebuie cautat? ";
				getline(cin, artist);
				cautarePieseArtist(prim, artist);
			}
			break;

		case 4:
			if (prim == nullptr) cout << "Nu exista piesa de sters.\n";
			else {
				cout << "Ce piesa trebuie stearsa? "; getline(cin, titlu);
				if (stergeT(prim, ultim, titlu)) {
					cout << "Piesa '" << titlu << "' a fost stearsa.\n";
				}
				else
					cout << "Nu exista piesa '" << titlu << "' care trebuie stearsa.\n";
			}
			break;

		case 5:
			if (prim == nullptr) cout << "Nu exista artist de sters.\n";
			else {
				cout << "Ce artist trebuie sa fie sters? "; getline(cin, artist);
				if (stergeA(prim, ultim, artist)) {
					cout << "Piesele artistului '" << artist << "' au fost sterse.\n";
				}
				else cout << "Nu exista artistul care trebuie sters.\n";
			}
			break;

		case 6:
			if (prim == nullptr) cout << "Nu exista piesa de adaugat la liked.\n";
			else {
				cout << "Ce piesa o adaugam la liked? "; getline(cin, titlu);
				Nod* piesaGasita = cautaT(prim, titlu);
				if (piesaGasita != nullptr) {
					piesaGasita->liked = true;
					cout << "Piesa '" << titlu << "' a fost adaugata la liked.\n";
				}
				else {
					cout << "Piesa '" << titlu << "' nu exista.\n";
				}
			}
			break;

		case 7:
			if (prim == nullptr) cout << "Nu exista piese ca sa facem un total.\n";
			else {
				double totalTimp = calcTimp(prim);
				cout << "Playlistul are: " << totalTimp << " minute.\n";
			}
			break;

		case 8:
			if (prim == nullptr) cout << "Nu exista piese in playlist.\n";
			else {
				cout << "Playlistul curent:\n";
				afiseaza(prim);
			}
			break;

		case 9:
			goleste(prim, ultim);
			cout << "Lista a fost golita.\n";
			break;

		case 0:
			cout << "La revedere!\n";
			break;

		default:
			cout << "Optiunea " << optiune << " este invalida\n";
		}
	} while (optiune != 0);

	goleste(prim, ultim);
	return 0;
}
